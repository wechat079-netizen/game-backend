const express = require('express'); const admin = require('firebase-admin'); const path = require('path'); const app = express(); app.use(express.json()); app.use(express.static(path.join(__dirname, 'public'))); // Firebase Admin SDK ကို Initialize လုပ်ခြင်း // (သင့်ရဲ့ Project ထဲမှာ serviceAccountKey.json ဖိုင် ထည့်ထားရပါမည်) const serviceAccount = require('./serviceAccountKey.json'); admin.initializeApp({ credential: admin.credential.cert(serviceAccount), databaseURL: "https://right-2c598-default-rtdb.firebaseio.com" }); const db = admin.database(); // ဂိမ်းအချက်အလက်များနှင့် History များ သိမ်းဆည်းရန် // Player က Bet တင်သည့် API app.post('/api/bet', async (req, res) => { try { const { gameType, playerName, choice, amount } = req.body; if (!gameType || !playerName || !choice || !amount) { return res.status(400).json({ success: false, message: 'အချက်အလက် မပြည့်စုံပါ။' }); } const betData = { playerName, choice, amount: Number(amount), timestamp: Date.now() }; // Firebase ထဲသို့ Player ရဲ့ My History နှင့် Current Bets များကို သိမ်းဆည်းမည် await db.ref(`games/${gameType}/currentBets`).push(betData); await db.ref(`players/${playerName}/history`).push({ gameType, choice, amount: Number(amount), timestamp: Date.now() }); res.json({ success: true, message: 'ထိုးကြေး အောင်မြင်ပါသည်။' }); } catch (error) { console.error("Bet Error:", error); res.status(500).json({ success: false, error: error.message }); } }); // Round ပြီးဆုံးသည့်အခါ Game History သိမ်းဆည်းရန် Function async function saveGameHistory(gameType, roundNumber, winningResult) { try { const historyRef = db.ref(`gameHistories/${gameType}`); await historyRef.push({ round: roundNumber, result: winningResult, finishedAt: new Date().toISOString() }); // လက်ရှိ Round ရဲ့ Bets များကို ရှင်းလင်းမည် await db.ref(`games/${gameType}/currentBets`).remove(); } catch (error) { console.error("History Save Error:", error); } } // Admin Panel အတွက် ဒေတာနှင့် History များ ဆွဲထုတ်ပေးသော API app.get('/api/admin/get-data', async (req, res) => { const adminToken = req.headers['x-admin-token']; const validPasswords = ['Peter125199', 'Win125199', '0084']; if (!validPasswords.includes(adminToken)) { return res.status(403).json({ error: 'Password မှားယွင်းနေပါသည်။' }); } const gameType = req.query.gameType || '30s'; try { // Firebase မှ လက်ရှိ Bets များကို ဖတ်ယူမည် const gameSnapshot = await db.ref(`games/${gameType}/currentBets`).once('value'); const betsObj = gameSnapshot.val() || {}; const bets = Object.keys(betsObj).map(key => ({ id: key, ...betsObj[key] })); // Total များကို တွက်ချက်ခြင်း let totals = { BIG: 0, SMALL: 0, GREEN: 0, VIOLET: 0 }; bets.forEach(b => { if (totals[b.choice] !== undefined) { totals[b.choice] += b.amount; } }); // Firebase မှ Game History များကို ဖတ်ယူမည် const historySnapshot = await db.ref(`gameHistories/${gameType}`).limitToLast(10).once('value'); const historyObj = historySnapshot.val() || {}; const histories = Object.keys(historyObj).map(key => historyObj[key]); res.json({ round: 1, // လိုအပ်ပါက သင့်ဂိမ်းတိုင်မာအလိုက် Round ထည့်ပါ timer: 30, forced: 'Auto', totals: totals, bets: bets, histories: histories }); } catch (error) { console.error("Admin Data Error:", error); res.status(500).json({ error: error.message }); } }); const PORT = process.env.PORT || 3000; app.listen(PORT, () => { console.log(`Server is running on port ${PORT}`); });
